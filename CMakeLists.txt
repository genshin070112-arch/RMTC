# 文件：CMakeLists.txt
# 作用：CMake 编译与安装配置
# 定义 C++17 核心库、ROS2 程序、五个 CTest 测试和独立裁判/ROS故障验收探针；仅安装参赛程序。
# 由 build.sh / colcon / Docker 构建调用；依赖 ROS2、libcurl、Threads 和随项目附带的 JSON 库。

cmake_minimum_required(VERSION 3.16)
project(rmtc_bot VERSION 1.1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(rmtc_core
  src/protocol.cpp src/events.cpp src/world.cpp src/planner.cpp
  src/controller.cpp src/prompts.cpp src/gateway.cpp src/config.cpp src/audit.cpp
  src/scoring.cpp src/tactics.cpp)
target_include_directories(rmtc_core PUBLIC include third_party)
target_link_libraries(rmtc_core PUBLIC CURL::libcurl Threads::Threads)
target_compile_options(rmtc_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rmtc_node src/node.cpp)
target_link_libraries(rmtc_node rmtc_core)
ament_target_dependencies(rmtc_node rclcpp std_msgs)
install(TARGETS rmtc_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_core.cpp")
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE rmtc_core)
  add_test(NAME rmtc_core_tests COMMAND test_core)
  add_executable(test_gateway tests/test_gateway.cpp)
  target_link_libraries(test_gateway PRIVATE rmtc_core)
  add_test(NAME rmtc_gateway_tests COMMAND test_gateway)
  add_executable(test_scoring tests/test_scoring.cpp)
  target_link_libraries(test_scoring PRIVATE rmtc_core)
  add_test(NAME rmtc_v11_scoring_tests COMMAND test_scoring)
  add_executable(test_review_rules tests/test_review_rules.cpp)
  target_link_libraries(test_review_rules PRIVATE rmtc_core)
  add_test(NAME rmtc_review_rules_tests COMMAND test_review_rules)
  add_executable(test_report tests/test_report.cpp)
  target_link_libraries(test_report PRIVATE rmtc_core)
  add_test(NAME rmtc_report_tests COMMAND test_report)
  add_executable(referee_probe tests/referee_probe.cpp)
  target_link_libraries(referee_probe rmtc_core)
  ament_target_dependencies(referee_probe rclcpp std_msgs)
  add_executable(ros_fault_probe tests/ros_fault_probe.cpp)
  target_link_libraries(ros_fault_probe rmtc_core)
  ament_target_dependencies(ros_fault_probe rclcpp std_msgs)
endif()
ament_package()
